package main

import (
	"os"
	"path/filepath"
	"testing"
	"time"

	"github.com/someone1/zfsbackup-go/files"
)

// Probe: a pipe-mode receive (--maxFileBuffer 0) of an intact small volume. Does it finish?
func TestAdvPipeReceiveSmallVolumeIntact(t *testing.T) {
	env := newE2EEnv(t)
	env.writeSnapshots(t, "tank/data", []files.SnapshotInfo{{Name: "a", CreationTime: time.Unix(100, 0)}})
	logs, err := env.send("--compressor", "", "tank/data@a", "file://"+env.dest)
	if err != nil {
		t.Fatalf("send: %v\n%s", err, logs)
	}
	receiveLog := filepath.Join(t.TempDir(), "receive.log")
	t.Setenv("FAKEZFS_RECEIVE_LOG", receiveLog)
	logs, err = guarded(t, func() (string, error) {
		return env.receive("--maxFileBuffer", "0", "tank/data@a", "file://"+env.dest, "restored/data")
	})
	if err != nil {
		t.Fatalf("receive: %v\n%s", err, logs)
	}
	if data, rerr := os.ReadFile(receiveLog); rerr != nil || len(data) == 0 {
		t.Errorf("zfs receive did not run: %v", rerr)
	}
}

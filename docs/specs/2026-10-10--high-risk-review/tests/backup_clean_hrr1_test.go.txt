// HRR Item 1: clean --cleanLocal and a cached local-only manifest of ANOTHER --manifestPrefix
// that this run cannot decode (truncated file, or encrypted to a key this run lacks).

// hrrSet is like cleanTestSet, but with an optional encryption key (and the EncryptTo the send
// would record, so the decoded manifest's StoredManifestObjectName matches the object name).
func hrrSet(t *testing.T, dir, dataset, snapshot, prefix string, key *openpgp.Entity, email string) (*files.JobInfo, string) {
	t.Helper()
	set := &files.JobInfo{
		VolumeName:     dataset,
		BaseSnapshot:   files.SnapshotInfo{Name: snapshot},
		ManifestPrefix: prefix,
		Separator:      "|",
		Compressor:     files.InternalCompressor,
		EncryptKey:     key,
		EncryptTo:      email,
	}
	vol := set.BackupVolumeObjectName(1)
	set.Volumes = []*files.VolumeInfo{{ObjectName: vol, VolumeNumber: 1}}
	writeTestManifest(t, dir, set)
	return set, writeTestObject(t, dir, vol)
}

// makeLocalOnly caches set's manifest (a clean under its prefix downloads it), then removes the
// manifest object from the destination so the cached copy is local-only. jobInfo's prefix and key
// are restored. It returns the cached file's path.
func makeLocalOnly(t *testing.T, targetDir string, jobInfo, set *files.JobInfo) string {
	t.Helper()
	savedPrefix, savedKey := jobInfo.ManifestPrefix, jobInfo.EncryptKey
	jobInfo.ManifestPrefix, jobInfo.EncryptKey = set.ManifestPrefix, set.EncryptKey
	if err := Clean(context.Background(), jobInfo, false, false); err != nil {
		t.Fatalf("clean under prefix %s (to cache its manifest): %v", set.ManifestPrefix, err)
	}
	jobInfo.ManifestPrefix, jobInfo.EncryptKey = savedPrefix, savedKey
	object := set.ManifestObjectName()
	if err := os.Remove(filepath.Join(targetDir, object)); err != nil {
		t.Fatal(err)
	}
	cache, err := getCacheDir(jobInfo, jobInfo.Destinations[0])
	if err != nil {
		t.Fatal(err)
	}
	path := filepath.Join(cache, cachedManifestName(object))
	if _, err = os.Stat(path); err != nil {
		t.Fatalf("expected the %s manifest in the cache: %v", set.ManifestPrefix, err)
	}
	return path
}

// truncateHalf cuts the file at path to half its size.
func truncateHalf(t *testing.T, path string) {
	t.Helper()
	info, err := os.Stat(path)
	if err != nil {
		t.Fatal(err)
	}
	if err = os.Truncate(path, info.Size()/2); err != nil {
		t.Fatal(err)
	}
	if _, err = readManifest(context.Background(), path, &files.JobInfo{}); err == nil {
		t.Fatalf("the truncated manifest %s still decodes; the test needs an undecodable one", path)
	}
}

// exists reports whether path exists (fatal on any other stat error).
func exists(t *testing.T, path string) bool {
	t.Helper()
	_, err := os.Stat(path)
	if err == nil {
		return true
	}
	if os.IsNotExist(err) {
		return false
	}
	t.Fatal(err)
	return false
}

// runCleanLocalBoth runs clean --cleanLocal as a dry run and then for real, and reports whether
// the dry run announced deleting the local manifest at cachedPath.
func runCleanLocalBoth(t *testing.T, jobInfo *files.JobInfo, cachedPath string) (dryLogs *bytes.Buffer) {
	t.Helper()
	dryLogs = captureLogs(t)
	if err := Clean(context.Background(), jobInfo, true, true); err != nil {
		t.Fatalf("dry run: %v", err)
	}
	if !exists(t, cachedPath) {
		t.Errorf("dry run deleted the cached manifest %s", cachedPath)
	}
	if err := Clean(context.Background(), jobInfo, true, false); err != nil {
		t.Fatalf("real run: %v", err)
	}
	return dryLogs
}

// Case 1: dataset A (this prefix) live; dataset B's manifest under prefix p2 is cached, gone from
// the destination, and truncated. Does clean --cleanLocal under the default prefix delete the
// cached p2 manifest? B's volume?
func TestHRR1CleanLocalTruncatedOtherPrefixOtherDataset(t *testing.T) {
	targetDir, jobInfo := setupCleanTest(t)
	hrrSet(t, targetDir, "tank/a", "a", "manifests", nil, "")
	setB, volB := hrrSet(t, targetDir, "tank/b", "a", "p2", nil, "")
	cached := makeLocalOnly(t, targetDir, jobInfo, setB)
	truncateHalf(t, cached)

	logs := runCleanLocalBoth(t, jobInfo, cached)
	if !strings.Contains(logs.String(), "Would delete local manifest "+cached) {
		t.Errorf("dry run did not announce deleting the other prefix's cached manifest:\n%s", logs.String())
	}
	if !exists(t, cached) {
		t.Errorf("clean --cleanLocal deleted the cached (truncated) manifest of prefix p2: %s", cached)
	}
	if !exists(t, volB) {
		t.Errorf("clean --cleanLocal deleted a volume of the p2 set: %s", volB)
	}
}

// Case 2: same as case 1 but both sets are of the SAME dataset tank/data (snapshot a under this
// prefix, snapshot b under p2).
func TestHRR1CleanLocalTruncatedOtherPrefixSameDataset(t *testing.T) {
	targetDir, jobInfo := setupCleanTest(t)
	hrrSet(t, targetDir, "tank/data", "a", "manifests", nil, "")
	setB, volB := hrrSet(t, targetDir, "tank/data", "b", "p2", nil, "")
	cached := makeLocalOnly(t, targetDir, jobInfo, setB)
	truncateHalf(t, cached)

	logs := runCleanLocalBoth(t, jobInfo, cached)

		t.Errorf("dry run did not announce deleting the other prefix's cached manifest:\n%s", logs.String())
	}
	if !strings.Contains(logs.String(), "Would delete "+joinURI(jobInfo.Destinations[0], setB.Volumes[0].ObjectName)) {
		t.Errorf("dry run did not announce deleting the p2 set's volume:\n%s", logs.String())
	}
	if !exists(t, cached) {
		t.Errorf("clean --cleanLocal deleted the cached (truncated) manifest of prefix p2: %s", cached)
	}
	if !exists(t, volB) {
		t.Errorf("clean --cleanLocal deleted a volume of the p2 set: %s", volB)
	}
}

// --- key case helpers (copied from files/pgp_test.go, which backup/ cannot import) ---

func hrrKey(t *testing.T, email string) *openpgp.Entity {
	t.Helper()
	e, err := openpgp.NewEntity("backup", "", email, &packet.Config{RSABits: 2048})
	if err != nil {
		t.Fatal(err)
	}
	for _, id := range e.Identities {
		id.SelfSignature.PreferredHash = []uint8{8}
		id.SelfSignature.PreferredSymmetric = []uint8{9}
	}
	if err = e.SerializePrivate(io.Discard, nil); err != nil {
		t.Fatal(err)
	}
	return e
}

func hrrLoadRings(t *testing.T, keys ...*openpgp.Entity) {
	t.Helper()
	dir := t.TempDir()
	for _, ring := range []struct {
		kind    string
		private bool
		load    func(string) error
	}{
		{openpgp.PublicKeyType, false, pgp.LoadPublicRing},
		{openpgp.PrivateKeyType, true, pgp.LoadPrivateRing},
	} {
		var buf bytes.Buffer
		w, err := armor.Encode(&buf, ring.kind, nil)
		if err != nil {
			t.Fatal(err)
		}
		for _, e := range keys {
			if ring.private {
				err = e.SerializePrivate(w, nil)
			} else {
				err = e.Serialize(w)
			}
			if err != nil {
				t.Fatal(err)
			}
		}
		w.Close()
		path := filepath.Join(dir, ring.kind)
		if err = os.WriteFile(path, buf.Bytes(), 0600); err != nil {
			t.Fatal(err)
		}
		if err = ring.load(path); err != nil {
			t.Fatal(err)
		}
	}
}

// Case 3: the p2 set (dataset tank/data@b) is encrypted to K2; this prefix's set (tank/data@a)
// to K1. The clean run holds K1 only. The cached p2 manifest cannot be decrypted: does
// --cleanLocal delete it and the p2 volume?
func TestHRR1CleanLocalWrongKeyOtherPrefix(t *testing.T) {
	targetDir, jobInfo := setupCleanTest(t)
	k1, k2 := hrrKey(t, "k1@example.com"), hrrKey(t, "k2@example.com")
	hrrLoadRings(t, k1, k2)
	t.Cleanup(func() { hrrLoadRings(t) }) // leave the package rings empty

	jobInfo.EncryptKey, jobInfo.EncryptTo = k1, "k1@example.com"
	hrrSet(t, targetDir, "tank/data", "a", "manifests", k1, "k1@example.com")
	setB, volB := hrrSet(t, targetDir, "tank/data", "b", "p2", k2, "k2@example.com")
	cached := makeLocalOnly(t, targetDir, jobInfo, setB)

	// The clean run has only K1.
	hrrLoadRings(t, k1)
	_, rerr := readManifest(context.Background(), cached, jobInfo)
	if rerr == nil {
		t.Fatal("the K2 manifest decodes with K1 only; the test setup is wrong")
	}
	t.Logf("readManifest with the wrong key: %T: %v (KeyError=%v)", rerr, rerr, errors.As(rerr, new(*files.KeyError)))

	logs := runCleanLocalBoth(t, jobInfo, cached)
	if !strings.Contains(logs.String(), "Would delete local manifest "+cached) {
		t.Errorf("dry run did not announce deleting the other prefix's cached manifest:\n%s", logs.String())
	}
	if !exists(t, cached) {
		t.Errorf("clean --cleanLocal deleted the cached manifest of prefix p2 (encrypted to another key): %s", cached)
	}
	if !exists(t, volB) {
		t.Errorf("clean --cleanLocal deleted a volume of the p2 set: %s", volB)
	}
}
